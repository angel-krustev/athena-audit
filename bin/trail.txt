aws s3api put-bucket-policy --bucket aws-logs-123456789-us-east-1 --policy '{
  "Version": "2012-10-17",
  "Statement": [
    {
      "Sid": "AWSCloudTrailAclCheck",
      "Effect": "Allow",
      "Principal": { "Service": "cloudtrail.amazonaws.com" },
      "Action": "s3:GetBucketAcl",
      "Resource": "arn:aws:s3:::aws-logs-123456789-us-east-1"
    },
    {
      "Sid": "AWSCloudTrailWrite",
      "Effect": "Allow",
      "Principal": { "Service": "cloudtrail.amazonaws.com" },
      "Action": "s3:PutObject",
      "Resource": "arn:aws:s3:::aws-logs-123456789-us-east-1/athena-cloudtrail/AWSLogs/123456789/*",
      "Condition": { "StringEquals": { "s3:x-amz-acl": "bucket-owner-full-control" } }
    }
  ]
}' --region us-east-1

aws cloudtrail create-trail --name athena-audit-trail --s3-bucket-name aws-logs-123456789-us-east-1 --s3-key-prefix athena-cloudtrail --is-multi-region-trail --region us-east-1 --no-cli-pager

aws cloudtrail start-logging --name athena-audit-trail --region us-east-1
